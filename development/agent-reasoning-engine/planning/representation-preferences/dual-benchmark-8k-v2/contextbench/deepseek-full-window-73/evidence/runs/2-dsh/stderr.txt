inline std::string prettify(const char *buf, size_t length) noexcept {
  // Heuristic to avoid excessive allocations.
  size_t margin = 32;
  size_t newlines = 0;
  for(size_t i = 0; i < length; i++) {
    newlines += (buf[i] == '{' || buf[i] == '[');
  }
  std::string out;
  out.reserve(length + margin + newlines * 4); // 4 spaces max per indent
  size_t indent = 0;
  bool quote = false;
  for(size_t i = 0; i < length; i++) {
    char c = buf[i];
    if(c == '"') quote = !quote;
    if (quote) { out.push_back(c); continue; }
    switch(c) {
    case '{':
    case '[':
      out.push_back(c);
      if(i + 1 < length && (buf[i+1] == '}' || buf[i+1] == ']')) {
        out.push_back(buf[i+1]); i++;

      out.push_back(c);
      out.push_back(' ');
      break;

      out.push_back(c);
      break;
    }
  }
