```cpp
  /**
   * @private This is the class used to pretty print. It produces
   * a version of the JSON with one value per line, indented with
   * four spaces per nesting level.
   */
  class simdjson_prettify_formatter final : public mini_formatter {
  public:
    /** Add a comma, followed by a newline and the indentation **/
    simdjson_inline void comma() {
      one_char(',');
      one_char('\n');
      print_indent();
    }
    /** Start an array, prints [ **/
    simdjson_inline void start_array() { one_char('['); depth++; }
    /** End an array, prints ] on its own line (or right after [ for empty arrays) **/
    simdjson_inline void end_array() { close_container(']'); }
    /** Start an object, prints { **/
    simdjson_inline void start_object() { one_char('{'); depth++; }
    /** End an object, prints } on its own line (or right after { for empty objects) **/
    simdjson_inline void end_object() { close_container('}'); }
    /** Prints a key (string + colon + space) **/
    simdjson_inline void key(std::string_view unescaped) {
      string(unescaped);
      one_char(':');
      one_char(' ');
    }
    /** Clears out the content. **/
    simdjson_inline void clear() { mini_formatter::clear(); depth = 0; }
  private:
    /** Ends a container, handling the empty-container case **/
    simdjson_inline void close_container(char c) {
      depth--;
      if(buffer.back() == char(c - 2)) { // container was empty: print [] or {}
        one_char(c);
      } else {
        one_char('\n');
        print_indent();
        one_char(c);
      }
    }
    /** Prints four spaces per nesting level **/
    simdjson_inline void print_indent() {
      for(size_t i = 0; i < depth; i++) {
        one_char(' '); one_char(' '); one_char(' '); one_char(' ');
      }
    }
    size_t depth{0};
  };
```
